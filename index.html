/* 저장 */


function saveMessages() {

    localStorage.setItem(

        STORAGE_KEY,

        JSON.stringify(messages)

    );

}


function saveSettings() {


    settings.scale = scale;

    settings.offsetX = offsetX;

    settings.offsetY = offsetY;


    localStorage.setItem(

        SETTINGS_KEY,

        JSON.stringify(settings)

    );

}



/* 시간 */


function formatTime(date) {


    return new Date(date)

        .toLocaleTimeString(

            "ko-KR",

            {

                hour: "numeric",

                minute: "2-digit"

            }

        );

}



/* 날짜 */


function dateString(date) {


    const d = new Date(date);


    return `${d.getFullYear()}.${String(

        d.getMonth() + 1

    ).padStart(2,"0")}.${String(

        d.getDate()

    ).padStart(2,"0")}`;

}



/* 렌더링 */


function renderMessages() {


    const keyword =

        search.value.trim().toLowerCase();


    messagesEl.innerHTML = "";


    let previousDate = "";


    messages

        .filter(message => {


            if (!keyword)

                return true;


            return message.text

                .toLowerCase()

                .includes(keyword);


        })

        .forEach(message => {


            const date =

                dateString(message.date);


            if (date !== previousDate) {


                const divider =

                    document.createElement("div");


                divider.className =

                    "date-divider";


                divider.innerHTML =

                    `<span class="date-label">${date}</span>`;


                messagesEl.appendChild(divider);


                previousDate = date;

            }


            const wrapper =

                document.createElement("div");


            wrapper.className =

                "message-wrapper";


            const content =

                document.createElement("div");


            content.className =

                "message-content";



            /* 사진 */


            if (message.image) {


                const img =

                    document.createElement("img");


                img.className =

                    "photo";


                img.src =

                    message.image;


                content.appendChild(img);

            }



            /* 텍스트 */


            if (message.text) {


                const bubble =

                    document.createElement("div");


                bubble.className =

                    "bubble";


                bubble.style.background =

                    settings.bubbleColor || "#111111";


                bubble.textContent =

                    message.text;


                content.appendChild(bubble);

            }



            /* 시간 */


            const time =

                document.createElement("div");


            time.className =

                "time";


            time.textContent =

                formatTime(message.date);


            content.appendChild(time);


            wrapper.appendChild(content);


            messagesEl.appendChild(wrapper);

        });


    messagesEl.scrollTop =

        messagesEl.scrollHeight;

}



/* 글 전송 */


function sendMessage() {


    const text =

        input.value.trim();


    if (!text)

        return;


    messages.push({

        id: crypto.randomUUID(),

        text: text,

        image: null,

        date: new Date().toISOString()

    });


    saveMessages();


    input.value = "";


    input.style.height = "40px";


    send.disabled = true;


    renderMessages();


    /*

     * 모바일 웹에서도 키보드가 내려가도록

     */

    input.blur();

}

    "pointerdown",

    event => {


        dragging = true;


        dragStartX =

            event.clientX -

            currentCropX;


        dragStartY =

            event.clientY -

            currentCropY;


        cropImage.setPointerCapture(

            event.pointerId

        );

    }

);



cropImage.addEventListener(

    "pointermove",

    event => {


        if (!dragging)

            return;


        currentCropX =

            event.clientX -

            dragStartX;


        currentCropY =

            event.clientY -

            dragStartY;


        updateCrop();

    }

);



cropImage.addEventListener(

    "pointerup",

    () => {


        dragging = false;

    }

);



/* 마우스 휠 확대/축소 */


cropImage.addEventListener(

    "wheel",

    event => {


        event.preventDefault();


        if (event.deltaY < 0) {


            currentCropScale =

                Math.min(

                    5,

                    currentCropScale + 0.1

                );


        } else {


            currentCropScale =

                Math.max(

                    1,

                    currentCropScale - 0.1

                );

        }


        updateCrop();

    },

    { passive: false }

);



/* 배경 조정 취소 */


document

    .getElementById("cancelCrop")

    .addEventListener(

        "click",

        () => {


            cropEl.classList.remove(

                "open"

            );

        }

    );



/* 배경 조정 저장 */


document

    .getElementById("saveCrop")

    .addEventListener(

        "click",

        () => {


            scale =

                currentCropScale;


            offsetX =

                currentCropX;


            offsetY =

                currentCropY;


            saveSettings();


            applyBackground();


            cropEl.classList.remove(

                "open"

            );

        }

    );



/* 초기 화면 */


renderMessages();
